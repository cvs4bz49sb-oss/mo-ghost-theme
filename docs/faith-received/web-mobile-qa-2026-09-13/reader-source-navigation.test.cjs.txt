function harness(href='https://reader.test/read?w=book#b1-0'){
  let now=0,serial=0;const timers=[],nodes=new Map(),placed=[],events=[],listeners={},hydrated=[],loads=[];
  const location=new URL(href),sc={scrollTop:0,clientHeight:600,visible:true,getClientRects(){return this.visible?[{}]:[];},getBoundingClientRect:()=>({top:64,height:600}),contains:()=>true};
  const reading={querySelectorAll:()=>[...nodes.values()].filter(n=>n.id?.startsWith('b'))},app={classList:{add(){},remove(){}}};
  nodes.set('scroll',sc);nodes.set('reading',reading);nodes.set('app',app);
  const schedule=(fn,delay=0)=>{timers.push({fn,time:now+delay,serial:++serial});};
  const window={__readerBuilt:true,dispatchEvent:e=>events.push(e),__ensurePage:page=>{hydrated.push(page);return true;}};
  const box={window,location,URL,URLSearchParams,CustomEvent:class{constructor(type,options){this.type=type;this.detail=options.detail;}},DATA:{slug:'book',pages:[]},cur:'1',
    history:{state:{kept:true},replaceState(state,title,url){this.state=state;location.href=String(url);}},
    document:{getElementById:id=>nodes.get(id),querySelectorAll:selector=>[...nodes.values()].filter(n=>selector.includes('data-source-path')?n.kind==='rhead':selector.includes('.pganchor')?n.kind==='pganchor':n.kind==='folio'),addEventListener:(name,fn)=>listeners[name]=fn},
    requestAnimationFrame:fn=>schedule(fn),setTimeout:schedule,setFolio(){},jump(page){loads.push(page);return null;}};
  vm.createContext(box);vm.runInContext(code,box);
  function row(id,{top=800,height=100,page=box.frReaderBlockReference(id)?.page||'1',kind='row'}={}){
    const node={id,kind,isConnected:true,dataset:{page},top,height,classList:{contains:cls=>cls===kind,add(){},remove(){}},
      getClientRects:()=>sc.visible?[{}]:[],getBoundingClientRect(){return {top:this.top-sc.scrollTop,bottom:this.top-sc.scrollTop+this.height,height:this.height};},
      querySelector(){return {textContent:this.textContent||''};},closest:selector=>selector==='.folio'?{dataset:{page}}:null,
      scrollIntoView({block}){placed.push(id);sc.scrollTop=this.top-(block==='start'?64:64+sc.clientHeight/2-this.height/2);}};
    nodes.set(id,node);if(!box.DATA.pages.some(p=>p.n===page))box.DATA.pages.push({n:page});return node;
  }
  function advance(ms=2000){const end=now+ms;let limit=2000;while(timers.length&&limit--){timers.sort((a,b)=>a.time-b.time||a.serial-b.serial);if(timers[0].time>end)break;const task=timers.shift();now=task.time;task.fn();}assert.ok(limit>0);now=end;}
  return {box,window,nodes,sc,location,placed,events,listeners,hydrated,loads,row,advance};
}

test('block parser preserves full encoded compound IDs and rejects malformed escapes',()=>{
  const h=harness(),r=h.box.frReaderBlockReference('#b52%3A0183A-7');
  assert.equal(r.id,'b52:0183A-7');assert.equal(r.page,'52:0183A');assert.equal(r.index,'7');
  assert.equal(h.box.frReaderBlockReference('#b%ZZ-0'),null);
});

test('accepted same-work navigation hydrates and lands an exact compound row without document navigation',()=>{
  const h=harness();h.row('b52:0183A-7');
  assert.equal(h.window.__frNavigateReaderAnchor('/read?w=book#b52%3A0183A-7'),true);h.advance();
  assert.deepEqual(h.placed,['b52:0183A-7']);assert.ok(h.hydrated.every(p=>p==='52:0183A'));assert.deepEqual(h.loads,[]);
  assert.equal(decodeURIComponent(h.location.hash),'#b52:0183A-7');assert.equal(h.box.history.state.kept,true);
  assert.equal(h.events.length,1);assert.equal(h.events[0].type,'fr-reader-navigation');
  assert.equal(h.events[0].detail.state,'landed');assert.equal(h.events[0].detail.page,'52:0183A');assert.equal(h.events[0].detail.id,'b52:0183A-7');
});
