  function directedPath(nodes,edges,start,end,direction='either'){
    const adjacency=nodes.map(()=>[]);for(const [a,b,w] of edges||[]){if(!adjacency[a]||!adjacency[b])continue;if(direction!=='in')adjacency[a].push([b,w]);if(direction!=='out')adjacency[b].push([a,w]);}

    while(head<queue.length){const v=queue[head++];if(v===end)break;for(const [n]of adjacency[v]||[])if(!previous.has(n)){previous.set(n,v);queue.push(n);}}
    if(!previous.has(end))return null;const path=[end];while(path.at(-1)!==start)path.push(previous.get(path.at(-1)));return path.reverse();
  }
  function journeyEdge(nodes,edges,left,right,direction='either',key=''){
    const matches=(edges||[]).filter(e=>nodes[e[0]]&&nodes[e[1]]&&((direction!=='in'&&e[0]===left&&e[1]===right)||(direction!=='out'&&e[0]===right&&e[1]===left)));
    return key?matches.find(e=>nodes[e[0]].s+'|'+nodes[e[1]].s===key)||null:matches.find(e=>e[0]===left)||matches[0]||null;
  }
  function journeyConnections(nodes,edges,current,direction='out'){
    const rows=[];
    for(const [from,to,total] of edges||[]){
      if(!nodes[from]||!nodes[to])continue;
      if(from===current&&direction!=='in')rows.push({node:nodes[to],citing:nodes[from],target:nodes[to],count:+total||0,direction:'out',key:nodes[from].s+'|'+nodes[to].s});
      if(to===current&&direction!=='out'&&from!==current)rows.push({node:nodes[from],citing:nodes[from],target:nodes[to],count:+total||0,direction:'in',key:nodes[from].s+'|'+nodes[to].s});
    }
    return rows.sort((a,b)=>b.count-a.count||a.node.a.localeCompare(b.node.a));
  }
