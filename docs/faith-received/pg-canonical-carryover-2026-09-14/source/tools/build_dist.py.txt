def _load(name, path):
    spec = importlib.util.spec_from_file_location(name, path)
    mod = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(mod)
    return mod

def _config_block(slug: str | None = None) -> str:
    """Canonical config <script> injected into every page."""
    lines = [
        f"window.__FR_BLOB_BASE__={json.dumps(BLOB_BASE)};",
        f"window.__FR_VER={json.dumps(BUILD_VER)};",   # cache-bust for mutable JSON (meta/work/shards/index/review)
        f"window.__FR_FB__={FB_JSON};",
        f"window.__FR_SITE__={json.dumps(SITE)};",
        "window.__FR_ASK_WORKSPACE__=true;",
        "try{const t=localStorage.getItem(\"fr_theme\");if([\"light\",\"dark\",\"sepia\"].includes(t))document.documentElement.dataset.theme=t;}catch(e){}",
    ]
    display_file = ROOT / "tools" / "visual_review" / "workgroups.json"
    if display_file.exists():
        display_groups = json.loads(display_file.read_text())
        display_volumes = {key: value["display_volume"] for key, value in display_groups.get("works", {}).items() if value.get("display_volume")}
        lines.append(f"window.__FR_EDITION_VOLUMES__={json.dumps(display_volumes)};")
    if slug is not None:
        lines.append(f"window.__FR_SLUG__={json.dumps(slug)};")
    return "<script>" + "".join(lines) + "</script>"


def _hoist_body_styles(html: str) -> str:
    """FIRST PAINT (owner 2026-09-10 "a magnifying glass briefly loads before the full site"): the library's own
    <style> blocks sat in the body ~250 KB after the markup, so the hero rendered unstyled — the search icon an
    unsized SVG at full width — until they streamed in. Move every body <style> block to the end of <head>, in
    their original order and AFTER the injected shared stylesheets, so the cascade is unchanged."""
    head_end = html.find("</head>")
    body_at = html.find("<body")
    if head_end < 0 or body_at < 0:
        return html
    blocks = re.findall(r"<style\b[^>]*>[\s\S]*?</style>", html[body_at:])
    if not blocks:
        return html
    body = html[body_at:]
    for blk in blocks:
        body = body.replace(blk, "", 1)
    return html[:head_end] + "".join(blocks) + html[head_end:body_at] + body

    html = html.replace('__RESEARCH_V__', BUILD_VER).replace('__TOOLS_V__', BUILD_VER)

# ── load reader template ──────────────────────────────────────────────────────
_reader_tpl_path = ROOT / "tools" / "prdl_reader_prototype" / "reader_shell.html"

def _package_reader_runtime(template: str) -> tuple[str, str]:
    """Keep the source self-contained, but share its unchanged classic script on deploy.

    The script remains at exactly the same document position. Its preload starts in
    the head, and its content hash stays stable across unrelated site releases.
    """
    matches = list(re.finditer(r'<script id="reader-core">([\s\S]*?)</script>', template))
    if len(matches) != 1:
        raise ValueError("reader package: expected exactly one reader-core script")
    match = matches[0]
    runtime = match[1]
    if len(runtime.encode("utf-8")) < 100_000:
        raise ValueError("reader package: core script unexpectedly small")
    version = hashlib.sha256(runtime.encode("utf-8")).hexdigest()[:16]
    url = f"/reader-core.js?v={version}"
    shell = template[:match.start()] + f'<script id="reader-core" src="{url}"></script>' + template[match.end():]
    shell = shell.replace("<head>", f'<head>\n<link rel="preload" as="script" href="{url}">', 1)
    if '<link rel="preload" as="script"' not in shell:
        raise ValueError("reader package: missing preload insertion point")
    return shell, runtime

def _stub_page(title, author, url, subtitle=""):
    return (f'<div class="tc"><div class=k>The Faith Received</div><h1>{title}</h1>'
            f'<p class=a>{author}{(" &middot; " + subtitle) if subtitle else ""}</p>'

def _seo_tags(work: dict) -> str:
    """Per-work meta tags for the SEO shell."""
    slug = work.get("slug", "")
    title = _html.escape(work.get("title") or slug)
    author = _html.escape(work.get("author") or "")
    n_pages = work.get("n_pages") or 0
    desc = _html.escape(
        f"{author}. {work.get('title', '')}. Latin–English parallel edition, {n_pages} pages."
    )
    url = _html.escape(f"{DOMAIN}/read/{slug}")
    # social card image: the work's title-page facsimile (Blob WebP). Scanned works only.
    tp = work.get("title_page") or 1
    og_img = (f'<meta property="og:image" content="{BLOB_BASE}/v1/works/{slug}/p/{tp}.webp">\n'
              f'<meta name="twitter:card" content="summary_large_image">\n') if work.get("has_pages") else \
             '<meta name="twitter:card" content="summary">\n'
    return (
        f'<title>{title} — The Faith Received</title>\n'
        f'<meta property="og:title" content="{title}">\n'
        f'<meta property="og:description" content="{desc}">\n'
        f'<meta name="description" content="{desc}">\n'
        f'<meta property="og:type" content="book">\n'
        f'<meta property="og:url" content="{url}">\n'
        f'<link rel="canonical" href="{url}">\n'
        + og_img
        # Highwire citation_* — the meta family Zotero (and Scholar) actually parses (10_RESEARCH_SYNTHESIS).
        # No citation_publication_date: library_meta has no year field (owner decision: ship without).
        + f'<meta name="citation_title" content="{title}">\n'
        + (f'<meta name="citation_author" content="{author}">\n' if author else '')
        + f'<meta name="citation_language" content="la">\n'
        f'<meta name="citation_public_url" content="{url}">\n'
    )


def _coins_span(work: dict) -> str:
    """COinS span (Zotero's most reliable detection channel — Princeton CDH 2025). Body-level.
    Book genre, no date field (no publication-year data yet)."""
    from urllib.parse import quote
    kev = "&".join([
        "ctx_ver=Z39.88-2004",
        "rft_val_fmt=" + quote("info:ofi/fmt:kev:mtx:book"),
        "rft.genre=book",
        "rft.btitle=" + quote(work.get("title") or work.get("slug", "")),
    ] + (["rft.au=" + quote(work["author"])] if work.get("author") else []) + [
        "rft.language=lat",
        "rft_id=" + quote(f"{DOMAIN}/read/{work.get('slug','')}"),
    ])
    return f'<span class="Z3988" title="{_html.escape(kev)}"></span>'


def _replace_head_title(html: str, new_tags: str) -> str:
    """Insert additional tags right after <head>; keep the template's own <title> as fallback
    but put the work-specific <title> first so it wins (browsers use the first one)."""
    # inject immediately after <head> open tag
    m = re.search(r"<head[^>]*?(?<!/)>", html, re.I)
    if m:
        pos = m.end()
        return html[:pos] + "\n" + new_tags + html[pos:]
    return new_tags + html


# ── build steps ───────────────────────────────────────────────────────────────
def build_read_html(out: pathlib.Path) -> None:
    """dist/read.html — reader template + config block. Also emit review.html (identical) so the
    legacy owner URL /review?ws=<slug> serves the reader natively (the static deploy has no /review
    route, and a vercel.json rewrite is shadowed by the edge gate middleware). The reader treats
    /review as a normal reader load in Blob mode — owner review = the masthead buttons + double-click."""
    html = _inject_config(READER_TEMPLATE, _config_block())
    html = html.replace("__TOOLS_V__", BUILD_VER)   # cache-bust the split tools bundle with the shell
    out.write_text(html, encoding="utf-8")
    (out.parent / "reader-core.js").write_text(READER_RUNTIME, encoding="utf-8")
    tools_src = ROOT / "tools" / "prdl_reader_prototype" / "read-tools.js"
    (out.parent / "read-tools.js").write_text(tools_src.read_text(encoding="utf-8"), encoding="utf-8")
    print(f"  wrote {out.relative_to(ROOT)}  ({out.stat().st_size:,} bytes)")
    review = out.with_name("review.html")
    review.write_text(html, encoding="utf-8")
    print(f"  wrote {review.relative_to(ROOT)}  (alias of read.html for /review?ws=)")


def build_index_html(out: pathlib.Path) -> None:
    """dist/index.html — library template with DESIGN substituted + config block."""
    html = LIBRARY_TEMPLATE.replace("__DESIGN__", DESIGN_CSS)
    html = _inject_config(html, _config_block())
    html = _hoist_body_styles(html)
    out.write_text(html, encoding="utf-8")
    print(f"  wrote {out.relative_to(ROOT)}  ({out.stat().st_size:,} bytes)")


def build_seo_shells(read_dir: pathlib.Path, works: list[dict]) -> int:
    """dist/read/{slug}.html — one SEO shell per work.

    FOLLOW-UP (owner 2026-08-17): the full-reader-copy shell (476KB × every work) is why
    a deploy hit 4.8GB — it duplicates the READER APP per slug, not content. PL works
    already ship as 1KB stubs (crawlable meta + canonical + redirect into /read?w=,
    which hydrates). EXTEND THE STUB PATH TO ALL BORN-DIGITAL WORKS (has_pages false):
    deploys drop to ~50MB, and the reader stays the single copy of itself. Facsimile
    works may keep full shells until their SEO impact is re-measured."""
    read_dir.mkdir(parents=True, exist_ok=True)
    count = 0
    for w in works:
        slug = w.get("slug")
        if not slug:
            continue
        # CANON-MODE works (PL, 8,967 of them): a full 476KB reader copy per work made the
        # deploy 4.8GB. They get a 1KB SEO stub — crawlable metadata + instant client
        # redirect into the query-param reader, which hydrates from the canon.
        # ALL BORN-DIGITAL works ship as ~550B SEO stubs (owner 2026-08-17 'go with the
        # stub'): crawlable metadata + canonical + instant redirect into the hydrating
        # reader. Only facsimile works (scan og:image) keep the full shell. Deploys drop
        # from ~780MB to ~50MB.
        if (not w.get("has_pages")) or slug.startswith("pg-") or slug.startswith("po-"):
            # canon works (PG/PO) are facsimile-backed but hydrate at read-time — stub them too
            if slug.startswith("eebo-"):
                # English Divines: the adapted EEBO reader hydrates from eebo/{id}.json.gz
                t = html_mod.escape(w.get("title") or slug); a = html_mod.escape(w.get("author") or "")
                url = f"/readen.html?id={slug[5:]}"
                (read_dir / f"{slug}.html").write_text(
                    f'<!doctype html><html lang="en"><head><meta charset="utf-8">'
                    f'<title>{t} — {a} — The Faith Received</title>'
                    f'<meta name="description" content="{t} by {a} ({html_mod.escape(str(w.get("volume") or ""))}). EEBO-TCP transcription, English Divines shelf.">'
                    f'<link rel="canonical" href="https://thefaithreceived.vercel.app{url}">'
                    f'<meta http-equiv="refresh" content="0;url={url}">'
                    f'<style>{STUB_CSS}</style>'
                    f'</head><body>{_stub_page(t, a, url, html_mod.escape(str(w.get("volume") or "")))}</body></html>',
                    encoding="utf-8")
                count += 1
                continue
            t = html_mod.escape(w.get("title") or slug); a = html_mod.escape(w.get("author") or "")
            vol = html_mod.escape(str(w.get("volume") or ""))
            desc = f"{t} by {a}." + (f" {vol}." if vol else "") + " The Faith Received digital library."
            (read_dir / f"{slug}.html").write_text(
                f'<!doctype html><html lang="en"><head><meta charset="utf-8">'
                f'<title>{t} — {a} — The Faith Received</title>'
                f'<meta name="description" content="{desc}">'
                f'<link rel="canonical" href="https://thefaithreceived.vercel.app/read?w={slug}">'
                f'<meta http-equiv="refresh" content="0;url=/read?w={slug}">'
                f'<style>{STUB_CSS}</style>'
                f'</head><body>{_stub_page(t, a, "/read?w=" + slug, vol)}</body></html>',
                encoding="utf-8")
            count += 1
            continue
        seo = _seo_tags(w)
        # Place SEO tags at the very start of <head>, then inject config before </head>
        html = _replace_head_title(READER_TEMPLATE, seo)
        html = _inject_config(html, _config_block(slug))
        # COinS lives in <body> (Zotero scans body for Z3988 spans)
        html = re.sub(r"(<body[^>]*?(?<!/)>)", lambda m: m.group(1) + _coins_span(w), html, count=1)
        dest = read_dir / f"{slug}.html"
        dest.write_text(html, encoding="utf-8")
        count += 1
    print(f"  wrote {count} SEO shells under dist/read/")
    return count


def build_sitemap(out: pathlib.Path, works: list[dict]) -> None:
    urls = "\n".join(
        f"  <url><loc>{DOMAIN}/read/{_html.escape(w['slug'])}</loc>"
        f"<lastmod>{TODAY}</lastmod></url>"
        for w in works
        if w.get("slug")
    )
    xml = f'<?xml version="1.0" encoding="UTF-8"?>\n<urlset xmlns="http://www.sitemaps.org/schemas/sitemap/0.9">\n{urls}\n</urlset>\n'
    out.write_text(xml, encoding="utf-8")
    print(f"  wrote {out.relative_to(ROOT)}  ({out.stat().st_size:,} bytes)")


def build_robots(out: pathlib.Path) -> None:
    out.write_text(
        f"User-agent: *\nAllow: /\nSitemap: {DOMAIN}/sitemap.xml\n",
        encoding="utf-8",
    )
    print(f"  wrote {out.relative_to(ROOT)}")

def build_css(out: pathlib.Path) -> None:
    """Dereference the symlink at tools/prdl_reader_prototype/prdl-system.css."""
    sym = ROOT / "tools" / "prdl_reader_prototype" / "prdl-system.css"
    try:
        real = sym.resolve(strict=True)
        shutil.copy2(str(real), str(out))
        print(f"  wrote {out.relative_to(ROOT)}  ({out.stat().st_size:,} bytes) [from {real}]")
    except (OSError, FileNotFoundError) as exc:
        # Never publish an empty design system during a File Provider outage.
        # This snapshot matches the production alias on 2026-09-13 byte for byte.
        # The canonical Dropbox symlink above remains the preferred source.
        import hashlib
        recovery = sym.with_name("prdl-system.recovery.css")
        expected = "773f88d15ebcac8e350408a207ff6e52df4f73bbe4ac95a80b5665ffe5a0dc8c"
        data = recovery.read_bytes()
        if hashlib.sha256(data).hexdigest() != expected:
            raise RuntimeError("Shared stylesheet unavailable and recovery checksum failed") from exc
        out.write_bytes(data)
        print(f"  WARN: canonical shared stylesheet unavailable ({exc}); used verified recovery snapshot ({len(data):,} bytes)")

_SECRET_PATTERNS = [
    "sk-or-",
    "vercel_blob_rw_",
    "vck_",
    "sk-proj-",
    "sk-ant-",
    "sk-cef",
    "BLOB_READ_WRITE",

def secrets_gate(dist: pathlib.Path) -> None:
    """Scan every file in dist/ for secret patterns; exit(1) on any hit.
    'AIza' (Firebase public web key) is explicitly allowed."""
    hits = []
    for f in sorted(dist.rglob("*")):
        if not f.is_file():
            continue
        try:
            text = f.read_text(encoding="utf-8", errors="replace")
        except Exception:
            continue
        for pat in _SECRET_PATTERNS:
            if pat in text:
                hits.append((str(f.relative_to(ROOT)), pat))
    if hits:
        for path, pat in hits:
            print(f"  SECRET LEAKED: {path}  pattern={pat!r}", file=sys.stderr)
        sys.exit(1)
    print("  secrets gate: PASS (no secrets found in dist/)")

def _page_gate(dist):
    """Abort if any inline <script> fails node --check, or a literal \\uXXXX escape sits in
    static HTML (renders literally in markup — the recurring escape-bug class)."""
    import re, subprocess, tempfile, sys
    from pathlib import Path
    core = Path(dist) / "reader-core.js"
    if not core.exists():
        sys.exit("page gate FAIL: packaged reader runtime is missing")
    checked = subprocess.run(["node", "--check", str(core)], capture_output=True, text=True)
    if checked.returncode:
        sys.exit(f"page gate FAIL: reader-core.js: {checked.stderr.strip()}")
    for html in sorted(Path(dist).glob("*.html")):
        src = html.read_text()
        # split on ANY opening <script ...> (incl. src= tags) + the close, so a src-only script's
        # empty body checks clean and the odd/even zone parity stays aligned.
        zones = re.split(r"<script\b[^>]*>|</script>", src)
        for i, z in enumerate(zones):
            if i % 2 == 1:
                with tempfile.NamedTemporaryFile("w", suffix=".js", delete=False) as f:
                    f.write(z); tmp = f.name
                r = subprocess.run(["node", "--check", tmp], capture_output=True, text=True)
                Path(tmp).unlink(missing_ok=True)
                if r.returncode != 0:
                    tail = (r.stderr or "").strip().splitlines()
                    sys.exit(f"page gate FAIL: {html.name} script#{(i+1)//2}: {tail[-1] if tail else 'syntax error'}")
            else:
                m = re.search(r"\\u[0-9a-fA-F]{4}", z)
                if m:
                    sys.exit(f"page gate FAIL: {html.name}: literal {m.group(0)} in static HTML — use an &entity;")
    print("  page gate: PASS (inline scripts and reader runtime parse; no literal escapes in markup)")
