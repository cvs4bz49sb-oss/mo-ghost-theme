const test=require('node:test'),assert=require('node:assert/strict'),vm=require('node:vm'),fs=require('node:fs');
const box={module:{exports:{}}};vm.runInNewContext(fs.readFileSync(__dirname+'/reader-navigation.js','utf8'),box);const N=box.module.exports;
test('numeric catalogue volumes cannot crash the reader library',()=>{const rows=N.catalogue({works:[{slug:'calov-1',title:'Systema',author:'Calov',volume:1},{slug:'bern',title:'Bern',author:'',volume:'1532'}]},{'calov-1':'System of theology'});assert.equal(rows.length,2);const c=rows.find(r=>r.slug==='calov-1');assert.equal(c.volume,'1');assert.equal(c.volume.replace(/&/g,'&amp;'),'1');assert.equal(c.title_en,'System of theology');});
test('mixed metadata is normalized and invalid catalogues remain retryable errors',()=>{const [r]=N.catalogue({works:[{slug:'test',title:12,author:['A','B'],volume:0},null,{}]},{test:{bad:true}});assert.equal(r.author,'A, B');assert.equal(r.title,'12');assert.equal(r.volume,'0');assert.equal(r.title_en,'');assert.throws(()=>N.catalogue({error:'unavailable'}));});
test('existing chapter and article labels are recognized without inventing headings',()=>{for(const t of ['Chapter one: We should administer our office with diligence','Chapter twenty-six: What should be reproved','Article IV. Of justification','Question 24: What is faith?'])assert.equal(N.chapterLabel(t),true,t);for(const t of ['The chapter discusses grace.','Chapter four is discussed in the introduction','A short sentence.','Chapter 1: '+ 'word '.repeat(90)])assert.equal(N.chapterLabel(t),false,t);});
test('confession navigation covers actual collection identifiers and their canonical slugs',()=>{for(const d of [{slug:'rc-010-bern-synod-1532'},{slug:'lc-002-augsburg-confession-1530'},{slug:'cf-001-apostolic-and-ancient-creeds'},{collection:'reformed-confessions'}])assert.equal(N.isConfession(d),true);assert.equal(N.isConfession({slug:'pg-230',title:'On confession'}),false);});
test('reader source remains parseable after navigation integration',()=>{const source=fs.readFileSync(__dirname+'/reader_shell.html','utf8');for(const m of source.matchAll(/<script\b[^>]*>([\s\S]*?)<\/script>/gi))if(m[1].trim())new vm.Script(m[1]);});

test('reader catalogue includes confessions without duplicating core works',()=>{const rows=N.catalogue({works:[{slug:'core',title:'Core',author:'A',volume:1}]},{},[{slug:'bern',title:'The Bern Synod',year:1532},{slug:'core',title:'Duplicate'}]);assert.equal(rows.length,2);assert.equal(rows.find(r=>r.slug==='bern').volume,'1532');assert.equal(rows.find(r=>r.slug==='bern')._confession,true);assert.equal(rows.find(r=>r.slug==='core').title,'Core');});

test('authored CHAP abbreviations accept Roman numbering and punctuation chains',()=>{for(const text of ['CHAP. I.—Of the Holy Scripture.','CHAP. XXXIII. — Of the Last Judgment.','Chap II: Of God.','§ CHAP. 3. - Of the decree.'])assert.equal(N.chapterLabel(text),true,text);for(const text of ['Chap. I is discussed here.','The Song of Songs','Examined and approved by the General Assembly'])assert.equal(N.chapterLabel(text),false,text);});

function sourceNode(title,id,tag='P',bold=false){const children=bold?[{tagName:'STRONG',textContent:title}]:[];return {id,tagName:tag,children,firstElementChild:children[0],parentElement:{children:[{}]},cloneNode:()=>({textContent:title,querySelectorAll:()=>[]}),closest:selector=>selector==='[data-page]'?{dataset:{page:'52:0183A'}}:null};}
test('Westminster-style source supplies all 33 chapters without Bible-list and approval-paragraph false positives',()=>{
 const roman=n=>{let out='';for(const [value,glyph]of [[10,'X'],[9,'IX'],[5,'V'],[4,'IV'],[1,'I']])while(n>=value){out+=glyph;n-=value;}return out;};
 const chapters=Array.from({length:33},(_,i)=>sourceNode('CHAP. '+roman(i+1)+'.—'+(i===0?'Of the Holy Scripture.':'Source chapter '+(i+1)+'.'),'b'+(i+1)+'-0'));
 const bible=['The Song of Songs','The Book of Psalms','The First Book of Kings'].map((title,i)=>sourceNode(title,'b1-'+(i+1),'P',i===1));
 const approval=sourceNode('Examined and approved by the General Assembly','b0-0');
 const input=[approval,chapters[0],...bible,...chapters.slice(1)],ids=input.map(node=>node.id),rows=N.contents({querySelectorAll:()=>input});
 assert.equal(rows.length,33);assert.deepEqual(Array.from(rows,r=>r.title),chapters.map(c=>c.cloneNode().textContent));assert.ok(rows.every(r=>r.depth===1));assert.deepEqual(input.map(node=>node.id),ids);assert.deepEqual(Array.from(rows,r=>r.anchor),chapters.map(c=>c.id));
});
test('genuine semantic subheadings remain in reading order under an authored chapter series',()=>{
 const input=[sourceNode('CHAP. I.—Of Scripture.','b1-0'),sourceNode('Proof from Scripture','b1-1','H4'),sourceNode('Article I. First proof.','b1-2','H4'),sourceNode('CHAP. II.—Of God.','b2-0')],rows=N.contents({querySelectorAll:()=>input});
 assert.deepEqual(Array.from(rows,r=>[r.title,r.depth]),[['CHAP. I.—Of Scripture.',1],['Proof from Scripture',2],['Article I. First proof.',2],['CHAP. II.—Of God.',1]]);assert.equal(rows[1].element,input[1]);
});
test('small unheaded confessions retain their existing fallback entries when no numbered series exists',()=>{
 const input=[sourceNode('The Faith We Confess','b52:0183A-0'),sourceNode('The Life of the Church','b52:0183A-1','P',true)],rows=N.contents({querySelectorAll:()=>input});assert.equal(rows.length,2);assert.deepEqual(Array.from(rows,r=>r.anchor),input.map(n=>n.id));assert.ok(rows.every(r=>r.page==='52:0183A'));
});

test('historical indexes group by source headings while preserving every opaque page ID',()=>{
 const pages=['0','52:0183A','52:0183B','201','202','203'].map(n=>({n}));
 const outline=[{page:'0',title:'Title'},{page:'52:0183A',title:'Index of authorities'},{page:'201',title:'Index of subjects'}];
 const result=N.indexSections([['52:0183A','alphabetic'],['52:0183B','citations'],['201','alphabetic'],['202','topical'],['203','topical']],outline,pages);
 assert.deepEqual(Array.from(result,g=>g.title),['Index of authorities','Index of subjects']);
 assert.deepEqual(Array.from(result[0].pages),['52:0183A','52:0183B']);assert.deepEqual(Array.from(result[1].pages),['201','202','203']);
});
test('unlabelled index pages remain available without invented letters or duplicate destinations',()=>{
 const result=N.indexSections([['0','alphabetic'],['0','alphabetic'],['appendix:A','alphabetic']],[],[{n:'0'},{n:'appendix:A'}]);
 assert.equal(result.length,1);assert.equal(result[0].title,'Index');assert.deepEqual(Array.from(result[0].pages),['0','appendix:A']);assert.equal(Object.hasOwn(result[0],'letters'),false);
});
