  const childLocation={href:loaded,replace(href){calls.replace.push(href);this.href=href;},reload(){calls.reload++;}};

    querySelector(selector){if(selector==='#scroll,.artscroll')return childScroll;if(selector==='#reading .row')return {};if(selector==='#reading')return {};return null;},addEventListener(){}};
  const child={location:childLocation,addEventListener(name,fn){childEvents.set(name,fn);},__frCaptureReaderPosition:()=>structuredClone(captured),__frCancelReaderNavigation:()=>calls.cancel++};
  if(childAPI){child.__frNavigateReaderAnchor=href=>{calls.navigate.push(href);childLocation.href=href;return true;};child.__frRestoreReaderPosition=position=>calls.restore.push(structuredClone(position));}

test('returning to the conversation cancels a source whose store update is still pending',async()=>{
  const h=harness({delayUpdates:true}),pending=h.api.open(url('alpha','9'),'Alpha',true,false,h.trigger);
  assert.equal(h.nodes.get('.fra-reader').hidden,false);
  h.api.show();h.resolve();await pending;h.flushFrames();
  assert.equal(h.nodes.get('.fra-reader').hidden,true);
  assert.equal(h.calls.replace.length,0);assert.equal(h.calls.navigate.length,0);
  assert.ok(h.calls.cancel>0);assert.equal(h.nodes.get('#fra-feed').scrollTop,432);
});

test('switching chats prevents delayed source completion from reopening the old reader',async()=>{
  const h=harness({delayUpdates:true}),pending=h.api.open(url('alpha','9'),'Alpha');
  const switching=h.api.switchChat('chat-b');await tick();
  assert.equal(h.api.snapshot().current,'chat-b');
  h.resolve(0);await pending;h.resolve(1);await switching;
  assert.equal(h.nodes.get('.fra-reader').hidden,true);
  assert.equal(h.api.snapshot().sourceURL,'');assert.equal(h.calls.replace.length,0);
  assert.equal(h.records[1].contextWork,'other','Old source does not alter the new conversation context');
});

test('latest source wins when an earlier store completion arrives last',async()=>{
  const h=harness({delayUpdates:true}),first=h.api.open(url('alpha','8'),'Alpha');
  const last=h.api.open(url('beta','52:0183A'),'Beta');
  h.resolve(1);await last;h.resolve(0);await first;
  assert.deepEqual(h.calls.replace,[url('beta','52:0183A')]);
  assert.equal(h.frame.dataset.source,url('beta','52:0183A'));
  assert.equal(h.nodes.get('#fra-source-title').textContent,'Beta');
  assert.equal(h.history.state.frAskSource.url,url('beta','52:0183A'));
});

  const target=origin+'/read?w=alpha&section=1&heading=chapter-two&p=5#b5-0';

test('explicit reclick navigates to the citation while resume restores the saved reading snapshot',async()=>{
  const target=url('alpha','52:0183A'),h=harness({loaded:target});
  const saved={id:'b52:0184B-2',page:'52:0184B',offset:-22,scroll:1780};h.api.seed(target,saved);
  await h.api.open(target,'Alpha',true,false,h.trigger);

test('delayed internal Back preserves a newer visible source and restores its history entry',async()=>{
  const h=harness(),first=url('alpha','4'),newer=url('beta','52:0183A');
  await h.api.open(first,'Alpha');h.api.back();
  assert.equal(h.calls.history.at(-1).type,'back');
  await h.api.open(newer,'Beta');const sequence=h.api.snapshot().sequence,cancels=h.calls.cancel;
  // The pending traversal now reaches the parent conversation entry.
  h.history.state={parentMarker:'conversation'};h.api.popstate();
  assert.equal(h.nodes.get('.fra-reader').hidden,false);
  assert.equal(h.api.snapshot().sourceURL,newer);assert.equal(h.api.snapshot().sequence,sequence);
  assert.equal(h.calls.cancel,cancels);assert.equal(h.calls.history.at(-1).type,'push');
  assert.equal(h.history.state.parentMarker,'conversation');assert.equal(h.history.state.frAskSource.url,newer);
  assert.equal(h.history.state.frAskSource.chat,'chat-a');assert.equal(h.history.state.frAskSource.title,'Beta');
  // One subsequent Back still returns from the new source normally.
  h.api.back();h.history.state={parentMarker:'conversation'};h.api.popstate();
  assert.equal(h.nodes.get('.fra-reader').hidden,true);
  assert.equal(h.calls.history.at(-1).type,'back');
});

test('delayed Back does not cancel a newer source still waiting for its store update',async()=>{
  const h=harness({delayUpdates:true}),first=h.api.open(url('alpha','4'),'Alpha');h.resolve(0);await first;
  h.api.back();const newer=url('beta','8'),pending=h.api.open(newer,'Beta');
  h.history.state=null;h.api.popstate();h.resolve(1);await pending;
  assert.equal(h.nodes.get('.fra-reader').hidden,false);
  assert.equal(h.calls.replace.at(-1),newer);assert.equal(h.history.state.frAskSource.url,newer);
});
