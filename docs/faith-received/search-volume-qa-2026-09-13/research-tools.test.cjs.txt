const {test}=require('node:test');const assert=require('node:assert/strict');const fs=require('node:fs'),vm=require('node:vm'),ctx={URL,module:{exports:{}}};vm.runInNewContext(fs.readFileSync(__dirname+'/research-tools.js','utf8'),ctx);const R=ctx.module.exports;
test('author grouping preserves shelf-specific routes without adding overlapping counts',()=>{const r=R.roster([{s:'john',a:'John',sh:'gf',w:969},{s:'john',a:'John',sh:'po',w:1}]);assert.equal(r.length,1);assert.equal(r[0].w,969);assert.equal(r[0].variants.map(x=>x.sh).join(','),'gf,po');assert.equal(R.authorURL(r[0],'Christ / Christology'),'/fathers?sh=gf#john/Christ%20%2F%20Christology');});
test('dates outrank unreliable era flags, missing dates stay undated',()=>{assert.equal(R.era({y:1483,e:'H'}),'R');assert.equal(R.era({e:'L'}),'U');assert.equal(R.era({y:1750}),'M');});
test('topic evidence includes people beyond the truncated author roster',()=>{const v=R.voices({authors:[{a:'A',s:'a'}],pos:[{a:'A',q:'first'},{a:'B',q:'second'}]});assert.equal(v.length,2);assert.equal(v[1].rows[0].q,'second');assert.equal(v[1].s,undefined);});
test('milestone strings and zero survive notebook serialization',()=>{assert.equal(R.page('52:0183A'),'52:0183A');assert.equal(R.page(0),'0');assert.equal(R.page(null),null);assert.equal(R.page(''),null);});
test('shared topic pages deduplicate repeats across exports without inferring missing pages',()=>{const row={w:'w',p:'52:0183A',x:['Faith']};const pairs=R.connections([{t:'Grace',pages:[row,row],pos:[row]},{t:'Faith',pages:[{w:'w',p:'52:0183A',x:['Grace']}]},{t:'Grace',pages:[{w:'x',x:['Faith']}]}]);assert.equal(pairs.length,1);assert.equal(pairs[0].pages.length,1);assert.equal(pairs[0].pages[0].p,'52:0183A');});
test('historical index reader links preserve exact anchors and reject off-site or active URLs',()=>{assert.equal(R.safeReaderURL('/read?w=pg-1#b52:0183A-0'),'/read?w=pg-1#b52:0183A-0');assert.equal(R.safeReaderURL('javascript:alert(1)'),null);assert.equal(R.safeReaderURL('https://example.com/read?w=x'),null);assert.equal(R.safeReaderURL('/read'),null);});
test('accent-insensitive discovery does not alter source labels',()=>assert.equal(R.fold('Suárez'),'suarez'));

test('anonymous shelf records and ambiguous topic metadata stay separate',()=>{assert.equal(R.roster([{s:'anonymous',sh:'pl'},{s:'anonymous',sh:'gf'}]).length,2);const v=R.voices({authors:[{a:'John',s:'john',sh:'gf',n:3},{a:'John',s:'john',sh:'po',n:4}],pos:[{a:'John',q:'text'}]});assert.equal(v[0].sh,null);assert.equal(v[0].n,null);assert.equal(v[0].rows.length,1);});

test('work graph references resolve only an unambiguous recorded Bible chapter',()=>{const context={window:{}};vm.runInNewContext(fs.readFileSync(__dirname+'/shelf-constellations.js','utf8'),context);const url=context.window.FRShelfMap.chapterURL,books=[{book:'Romans',slug:'romans',chapters:[{c:8}]},{book:'I Corinthians',slug:'i-corinthians',chapters:[{c:15}]},{book:'John',slug:'john',chapters:[{c:3}]},{book:'Job',slug:'job',chapters:[{c:3}]}];assert.equal(url('Rom. 8',books),'/bible#b/romans/8');assert.equal(url('1 Cor. 15',books),'/bible#b/i-corinthians/15');assert.equal(url('Jo. 3',books),null);assert.equal(url('Rom. 99',books),null);});

test('inline previews keep exact source routes and exclude external destinations',()=>{const context={URL,window:{}};vm.runInNewContext(fs.readFileSync(__dirname+'/shelf-constellations.js','utf8'),context);const url=context.window.FRShelfMap.previewURL;assert.equal(url('/read?w=pg-1#b52:0183A-0'),'/read?w=pg-1#b52:0183A-0');assert.equal(url('/read?w=w#b0-0'),'/read?w=w#b0-0');assert.equal(url('/bible#b/romans/8'),'/bible#b/romans/8');assert.equal(url('https://example.com/read?w=w'),null);assert.equal(url('javascript:alert(1)'),null);});

test('extraction notes stay distinct from legitimate specialist topics',()=>{for(const label of ['Repentance is not listed','Fear of God (not on list)','Adoption is not a locus; use closest: Grace','Marriage is not in the closed list','loci_other'])assert.equal(R.isRawTopic(label),true,label);for(const label of ['Repentance','Fear of God','Canon Law','Grace','Marriage and Divorce'])assert.equal(R.isRawTopic(label),false,label);});

test('connection summaries exclude raw labels while retaining legitimate shared subjects',()=>{const rows=R.connections([{t:'Grace',pages:[{w:'w',p:0,x:['Faith','Repentance is not listed']}]}]);assert.equal(rows.length,1);assert.equal(rows[0].a,'Faith');assert.equal(rows[0].b,'Grace');assert.equal(rows[0].pages[0].p,0);});

test('patristic citations accept Arabic and Roman volume numbers without prefix matching',()=>{assert.equal(R.seriesRef('PL XXXII Augustine').volume,32);assert.equal(R.seriesRef('Patrologia Graeca 64').series,'PG');assert.equal(R.seriesRef('PO Tome II').volume,2);assert.equal(R.seriesRef('PL IIII'),null);assert.equal(R.seriesRef('PL 32').volume===R.seriesRef('PL 320').volume,false);});
test('edition labels preserve column identifiers and do not invent PO column ranges',()=>{assert.equal(R.edition({volume:'PG 52',cols:['0183A','0184D']}),'PG 52, cols. 0183A–0184D');assert.equal(R.edition({volume:'PL 32',cols:[59,59]}),'PL 32, col. 59');assert.equal(R.edition({volume:'PO Tome 2',cols:[1,80]}),'PO 2');assert.equal(R.edition({volume:'Volume IV (1672)'}),'Volume IV (1672)');assert.equal(R.edition({}),'');});
test('works follow numeric volume and column order while distinct records remain separate',()=>{const rows=[{w:'pg-3',t:'Letters',volume:'PG 10',cols:['0183A','0184D']},{w:'pg-2',t:'Letters',volume:'PG 2',cols:['0184A','0185D']},{w:'pg-1',t:'Letters',volume:'PG 2',cols:['0183A','0183D']}].sort(R.workOrder);assert.equal(rows.map(w=>w.w).join(','),'pg-1,pg-2,pg-3');});
test('recorded Roman-numbered volumes keep their printed labels and sort numerically',()=>{const rows=[{volume:'Volume IX'},{volume:'Volume IV'},{volume:'Volume V'}].sort(R.workOrder);assert.equal(rows.map(R.edition).join('|'),'Volume IV|Volume V|Volume IX');});
test('a shelf topic count uses the published page label, never a graph weight',()=>{const rows=R.scopedTopics([{t:'Excommunication',s:'excommunication',n:58,na:28},{t:'Grace',n:90}],[{a:'Excommunication',n:169,sub:'6 mined pages'}]);assert.equal(rows.length,1);assert.equal(rows[0].n,6);assert.equal(rows[0].na,null);assert.equal(R.scopedTopics([{t:'Grace'}],[{a:'Grace',n:99}])[0].n,null);});
test('page references combine room and map samples without truncating columns or inventing links',()=>{const rows=R.mergePageRefs([{w:'w',p:'52:0183A',g:'short'},{w:'z',p:0}],[{w:'w',p:'52:0183A',g:'longer summary'},{w:'w',p:'52:0183B'},{w:'missing'}]);assert.equal(rows.length,3);assert.equal(rows[0].p,'52:0183A');assert.equal(rows[0].g,'longer summary');assert.equal(rows[1].p,'0');});
test('map references require an exact topic and author prefix',()=>{const nodes=[{a:'Grace',rows:[{t:'John Smith · Work',w:'a',p:1},{t:'John · Work',w:'b',p:2}]}];assert.equal(R.mapAuthorPages(nodes,'Grace','John').length,1);assert.equal(R.mapAuthorPages(nodes,'Grace','John')[0].w,'b');assert.equal(R.mapAuthorPages(nodes,'Graces','John').length,0);});
test('topic routes retain the selected shelf and reject unrecognized scope',()=>{assert.equal(R.topicURL('excommunication','md'),'/topics?sh=md#excommunication');assert.equal(R.topicURL('','md'),'/topics?sh=md');assert.equal(R.topicURL('god','bad'),'/topics#god');});

test('author matching accepts reordered names, accents, variants and accidental spaces',()=>{
 const author={a:'Thomas Aquinas',s:'thomas-aquinas',variants:[{a:'Thomas de Aquino'}]};
 for(const query of ['aquinas thomas','THOMAS','aqu inas','thom aqu','de aquino'])assert.ok(R.authorScore(author,query)>0,query);
 assert.ok(R.authorScore({a:'Francisco Suárez'},'francisco suarez')>0);assert.equal(R.authorScore(author,'Bonaventure'),0);
 assert.ok(R.authorScore({a:'John Owen'},'John Owen')>R.authorScore({a:'John Owens'},'John Owen'));
 assert.equal(R.authorScore({a:'John',variants:[]},'Richard Baxter'),0);
});
