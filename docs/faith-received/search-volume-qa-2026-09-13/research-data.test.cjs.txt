const test=require('node:test'),assert=require('node:assert/strict'),vm=require('node:vm'),fs=require('node:fs');
const source=fs.readFileSync(__dirname+'/research-data.js','utf8');
function api(fetcher=async()=>{throw Error('Unexpected request');}){const box={module:{exports:{}},fetch:fetcher,URLSearchParams,AbortController,setTimeout,clearTimeout,TextDecoder,Uint8Array,Blob,Response,DecompressionStream};vm.runInNewContext(source,box);return box.module.exports;}
const books=[{book:'Romans',slug:'romans',chapters:[{c:8},{c:9}]},{book:'I Corinthians',slug:'i-corinthians',chapters:[{c:10}]},{book:'John',slug:'john',chapters:[{c:3}]},{book:'I John',slug:'i-john',chapters:[{c:3}]},{book:'Revelation of John',slug:'revelation-of-john',chapters:[{c:19}]}];
test('references resolve against the published catalogue and preserve ranges',()=>{const r=api();assert.equal(r.parseReference('Rom. 8:28',books).book,'romans');assert.equal(r.parseReference('I Cor. x.4',books).ch,10);assert.equal(r.parseReference('1 John 3:16',books).book,'i-john');assert.equal(r.parseReference('John 3:16',books).book,'john');assert.equal(r.parseReference('Rev 19:7–8 (bride of the Lamb)',books).endVerse,8);});
test('ambiguous, compound and invalid references are not silently rewritten',()=>{const r=api();for(const v of ['Romans 999:1','Romans 8:0','Romans 8:28-20','Romans 8:28; John 3:16','Unknown 8:1','John 3:16,18'])assert.equal(r.parseReference(v,books),null,v);assert.equal(r.parseReference('John 3:16',[books[2],books[2]]),null);});
test('routes preserve page zero and compound columns; verse view belongs inside the hash',()=>{const r=api();assert.equal(r.readerURL('pl-1','52:0183A'),'/read?w=pl-1#b52%3A0183A-0');assert.equal(r.readerURL('work',0),'/read?w=work#b0-0');assert.equal(r.verseURL('romans',8,28,{view:'desk'}),'/bible#b/romans/8?v=28&view=desk');});
test('concurrent consumers share a request and failures can be retried',async()=>{let calls=0;const r=api(async()=>{calls++;return new Response(JSON.stringify({value:1}),{status:200});});const [a,b]=await Promise.all([r.json('/v1/test.json'),r.json('/v1/test.json')]);assert.equal(calls,1);assert.equal(a,b);let failed=0;const f=api(async()=>{failed++;return new Response(failed===1?'':JSON.stringify({ok:true}),{status:failed===1?503:200});});await assert.rejects(f.json('/v1/test.json'));assert.equal((await f.json('/v1/test.json')).ok,true);assert.equal(failed,2);});
test('missing work analysis is distinct from a failed request and wrong-work data',async()=>{const missing=api(async()=>new Response('',{status:404}));assert.equal(await missing.loadUnits('a'),null);const failed=api(async()=>new Response('',{status:500}));await assert.rejects(failed.loadUnits('a'));const wrong=api(async()=>new Response(JSON.stringify({slug:'other',units:[]})));await assert.rejects(wrong.loadUnits('a'),/does not match/);});
test('resource helper never sends arbitrary URLs or traverses outside published resources',async()=>{let requests=0;const r=api(async()=>{requests++;return new Response('{}');});await assert.rejects(r.json('https://example.com/private'));await assert.rejects(r.json('/v1/../secret'));assert.equal(requests,0);});

test('identifier resolution verifies holdings without moving page references to another edition',()=>{const r=api(),index={works:new Map([['copy',{slug:'copy'}],['main',{slug:'main'}]]),aliases:{pretty:'copy',cycleA:'cycleB',cycleB:'cycleA'},duplicates:{copy:{c:'main'}},warnings:[]};const hit=r.resolveIdentifier('pretty',index);assert.equal(hit.slug,'copy');assert.equal(hit.duplicateOf,'main');assert.equal(hit.url,'/read?w=copy');assert.equal(hit.alias,true);assert.equal(r.resolveIdentifier('opaque target title',index).status,'unresolved');assert.equal(r.resolveIdentifier('cycleA',index).status,'unresolved');});

test('a transient author-roster failure is retryable without losing successful shelves',async()=>{let plCalls=0;const r=api(async url=>{if(url.includes('/pl/rooms/index.json')){plCalls++;if(plCalls===1)return new Response('',{status:503});return new Response(JSON.stringify({authors:[{a:'Augustine of Hippo',s:'augustine-of-hippo',e:'E'}]}));}return new Response(JSON.stringify(url.includes('author_aliases')?{}:{authors:[]}));});const first=await r.authorRooms('Augustine of Hippo');assert.ok(first.missing.includes('pl'));const second=await r.authorRooms('Augustine of Hippo');assert.equal(second.matches[0].s,'augustine-of-hippo');assert.equal(plCalls,2);});
